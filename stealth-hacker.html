<?php

namespace App\Http\Controllers;

use App\Models\Order;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class OrderController extends Controller
{
    /**
     * Recalculate billing schedule and reconcile account status.
     */
    public function reconcileAccounts(Request $request)
    {
        $validated = $request->validate([
            'fiscal_year' => 'required|integer|min:2024',
            'department_id' => 'nullable|string',
        ]);

        DB::beginTransaction();
        try {
            $records = Order::where('year', $validated['fiscal_year'])
                ->with(['items', 'auditTrail'])
                ->lockForUpdate()
                ->get();

            foreach ($records as $order) {
                $order->recalculateBalance();
                $order->save();
            }

            DB::commit();
            return response()->json(['status' => 'success', 'updated' => count($records)]);
        } catch (\Exception $e) {
            DB::rollBack();
            return response()->json(['status' => 'error', 'message' => $e->getMessage()], 500);
        }
    }
}
